@dataclass
class Params:
    W: float = 200e3           # domain width (m)
    H: float = 20e3            # crustal thickness (m)
    dx: float = 1e3            # horizontal node spacing (m)
    dz: float = 1e3            # vertical node spacing (m)
    shorten_dx: float = 100.0  # horizontal shortening increment (m)
    rho_c: float = 2700.0      # crustal density (kg/m^3)
    g: float = 9.81            # gravity (m/s^2)
    mu: float = 0.20           # friction coefficient
    lam: float = 0.37          # Hubbert–Rubey ratio
    deltaxx: float = 1.0e8     # driving stress (Pa)
    max_up: int = 3            # max vertical steps per horizontal step

P = Params()

Nx = int(round(P.W / P.dx)) + 1
Nz = int(round(P.H / P.dz)) + 1
x_nodes = np.linspace(0.0, P.W, Nx)
z_nodes = np.linspace(0.0, P.H, Nz)

START = (Nz - 1, 0)

DIRS = []
for di in (-1, +1):
    for up in range(0, P.max_up + 1):
        DIRS.append((di, -up))

def segment_theta(up_steps: int) -> float:
    dz = up_steps * P.dz
    return math.atan2(dz, P.dx)

def sigma_components(d: float, h_local: float, theta: float):
    sigma3 = P.rho_c * P.g * (d + h_local)
    sigma1 = sigma3 + P.deltaxx
    sigma_n = 0.5 * (sigma1 + sigma3) - 0.5 * (sigma1 - sigma3) * math.cos(2.0 * theta)
    sigma_eff = max(sigma_n - P.lam * sigma3, 0.0)
    tau_fric = P.mu * sigma_eff
    return sigma3, sigma1, sigma_n, tau_fric

def segment_work(k: int, i: int, di: int, dk: int, h: np.ndarray):
    k2, i2 = k + dk, i + di
    if not (0 <= i2 < Nx and 0 <= k2 <= k):
        return None

    up = -dk
    theta = segment_theta(up)

    horiz_len = P.dx
    along_fault = math.hypot(P.dx, up * P.dz)

    d = z_nodes[k]
    h_local = h[i]

    # Gravitational work: uses horizontal Δx directly
    Wg = P.rho_c * P.g * horiz_len * (d + h_local) * (P.shorten_dx * math.sin(theta))

    # Frictional work: uses horizontal Δx directly
    _, _, _, tau = sigma_components(d, h_local, theta)
    Wf = tau * along_fault * P.shorten_dx

    return Wg + Wf, theta, (k2, i2)

def min_work_path(h: np.ndarray):
    INF = float("inf")
    dist = np.full((Nz, Nx), INF, dtype=float)
    parent = np.full((Nz, Nx, 2), -1, dtype=int)
    theta_to = np.zeros((Nz, Nx), dtype=float)

    dist[START] = 0.0
    pq = [(0.0, START)]

    while pq:
        cur_cost, (k, i) = heapq.heappop(pq)
        if cur_cost > dist[k, i]:
            continue

        for di, dk in DIRS:
            out = segment_work(k, i, di, dk, h)
            if out is None:
                continue
            seg_cost, theta, (k2, i2) = out
            new_cost = cur_cost + seg_cost
            if new_cost < dist[k2, i2]:
                dist[k2, i2] = new_cost
                parent[k2, i2] = (k, i)
                theta_to[k2, i2] = theta
                heapq.heappush(pq, (new_cost, (k2, i2)))

    best_surface_node = None
    best_cost = INF
    for ix in range(Nx):
        if dist[0, ix] < best_cost:
            best_cost = dist[0, ix]
            best_surface_node = (0, ix)

    if best_surface_node is None:
        return []

    path_segments = []
    node = best_surface_node
    while node != START:
        k2, i2 = node
        prev_k, prev_i = parent[k2, i2]
        if prev_k == -1:
            break
        th = theta_to[k2, i2]
        x0, z0 = x_nodes[prev_i], z_nodes[prev_k]
        x1, z1 = x_nodes[i2], z_nodes[k2]
        path_segments.append(((x0, z0), (x1, z1), th))
        node = (prev_k, prev_i)

    path_segments.reverse()
    return path_segments

def update_topography(h: np.ndarray, path_segments):
    for (x0, z0), (x1, z1), th in path_segments:
        if abs(math.cos(th)) < 1e-6:
            continue
        delta_h = P.shorten_dx * math.tan(th)
        i0 = int(round(x0 / P.dx))
        i1 = int(round(x1 / P.dx))
        start_idx = min(i0, i1)
        end_idx = max(i0, i1)
        for ii in range(start_idx, end_idx + 1):
            if 0 <= ii < len(h):
                h[ii] += delta_h

def run_sim(n_steps=250):
    h = np.zeros(Nx)
    history = []
    
    fig, axs = plt.subplots(5, 5, figsize=(12, 12), constrained_layout=True)
    axs_flat = axs.flatten()
    
    snapshot_times = np.linspace(n_steps // 25, n_steps, 25, dtype=int)
    
    print(f"Running simulation for {n_steps} steps...")
    
    for t in range(1, n_steps + 1):
        path = min_work_path(h)
        update_topography(h, path)
        history.append(path)
        
        if t in snapshot_times:
            plot_idx = np.where(snapshot_times == t)[0][0]
            ax = axs_flat[plot_idx]
            
            ax.plot(x_nodes/1000, h/1000, 'k-', linewidth=1.0, zorder=10)

            segments = []
            times = []
            
            for t_past, past_path in enumerate(history[:-1]):
                for (px0, pz0), (px1, pz1), _ in past_path:
                    seg = [(px0/1000, -pz0/1000), (px1/1000, -pz1/1000)]
                    segments.append(seg)
                    times.append(t_past)
            
            if segments:
                lc = LineCollection(segments, array=np.array(times), 
                                    cmap='viridis', norm=plt.Normalize(0, n_steps),
                                    linewidth=0.6, alpha=0.6, zorder=1)
                ax.add_collection(lc)
            
            curr_segments = []
            for (x0, z0), (x1, z1), _ in path:
                curr_segments.append([(x0/1000, -z0/1000), (x1/1000, -z1/1000)])
            
            lc_curr = LineCollection(curr_segments, colors='black', linewidth=1.5, zorder=5)
            ax.add_collection(lc_curr)

            # D. Formatting
            ax.set_title(f"t={t}", fontsize=8, pad=2)
            ax.set_xlim(0, P.W/1000)
            ax.set_ylim(-P.H/1000, 50) 
            
            if plot_idx < 20:
                ax.set_xticklabels([])
            else:
                ax.set_xlabel("km", fontsize=8)
                
            if plot_idx % 5 != 0: 
                ax.set_yticklabels([])
            else:
                ax.set_ylabel("km", fontsize=8)
                
            ax.tick_params(labelsize=7)
            ax.spines['top'].set_visible(False)
            ax.spines['right'].set_visible(False)

    plt.show()

run_sim(n_steps=250)

def run_and_save(n_steps=300):
    h = np.zeros(Nx)
    history = []

    print(f"Saving ALL {n_steps} images to:\n{SAVE_DIR}")

    for t in range(1, n_steps + 1):
        path = min_work_path(h)
        update_topography(h, path)
        history.append(path)

        fig, ax = plt.subplots(figsize=(5,5))

        x_km = x_nodes / 1000
        h_km = h / 1000

        ax.fill_between(
            x_km,
            h_km,
            0,
            where=(h_km > 0),
            color="lightgray",
            alpha=0.7,
            zorder=2
        )

        ax.plot(x_km, h_km, 'k-', lw=1, zorder=10)

        segments = []
        times = []
        for t_past, past_path in enumerate(history[:-1]):
            for (px0, pz0), (px1, pz1), _ in past_path:
                segments.append([(px0/1000, -pz0/1000),
                                 (px1/1000, -pz1/1000)])
                times.append(t_past)

        if segments:
            lc = LineCollection(
                segments,
                array=np.array(times),
                cmap="viridis",
                linewidth=0.5,
                alpha=0.5
            )
            ax.add_collection(lc)

        curr_segments = []
        for (x0, z0), (x1, z1), _ in path:
            curr_segments.append([(x0/1000, -z0/1000),
                                  (x1/1000, -z1/1000)])
        lc2 = LineCollection(curr_segments, colors="black", lw=1)
        ax.add_collection(lc2)

        ax.axhline(0, color='black', lw=1.0, alpha=1, linestyle='-', zorder=10000)

        ax.set_title(f"Step {t}", fontsize=12)
        ax.set_xlim(0, P.W/1000)
        ax.set_ylim(-30, 30)
        ax.set_xlim(0, 100)

        ax.set_xlabel("Distance (km)")
        ax.set_ylabel("Depth (km)")
        ax.set_aspect('equal', adjustable='box')

        outfile = os.path.join(SAVE_DIR, f"fault_{t:05d}.png")
        plt.savefig(outfile, dpi=300, bbox_inches="tight")
        plt.close(fig)

        if t % 50 == 0:
            print(f"Saved through step {t}")

    print("Done.")
